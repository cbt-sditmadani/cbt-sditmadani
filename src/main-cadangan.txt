import React, { useEffect, useMemo, useState } from 'react';
import { createRoot } from 'react-dom/client';
import { configured, supabase } from './supabase';
import { SCHOOL_NAME, SCHOOL_LOGO_URL } from './config';
import './style.css';

function App(){
  const [page,setPage]=useState('home');
  const [msg,setMsg]=useState('');
  const [busy,setBusy]=useState(false);
  const [session,setSession]=useState(null);
  const [exam,setExam]=useState(null);
  const [questions,setQuestions]=useState([]);
  const [answers,setAnswers]=useState({});
  const [idx,setIdx]=useState(0);
  const [student,setStudent]=useState({name:'',class_name:'',code:''});
  const [remaining,setRemaining]=useState(0);
  const [result,setResult]=useState(null);
  const [adminEmail,setAdminEmail]=useState('');
  const [adminPassword,setAdminPassword]=useState('');
  const [exams,setExams]=useState([]);
  const [newExam,setNewExam]=useState({title:'',code:'',duration_minutes:60});
  const [editingExam,setEditingExam]=useState(null);
  const [draft,setDraft]=useState({type:'multiple_choice',prompt:'',options:['','','',''],correct_answer:'',points:1});
  const [examQuestions,setExamQuestions]=useState([]);
  const [gradeRows,setGradeRows]=useState([]);

  useEffect(()=>{ if(!supabase)return; supabase.auth.getSession().then(({data})=>setSession(data.session)); const {data:{subscription}}=supabase.auth.onAuthStateChange((_e,s)=>setSession(s)); return ()=>subscription.unsubscribe();},[]);
  useEffect(()=>{ if(session){setPage('admin');loadExams();} },[session]);
  useEffect(()=>{ if(page!=='taking')return; const timer=setInterval(()=>setRemaining(v=>{if(v<=1){clearInterval(timer);finishExam(true);return 0;}return v-1;}),1000);return ()=>clearInterval(timer);},[page]);
  const current=questions[idx];
  const fmt=s=>`${String(Math.floor(s/60)).padStart(2,'0')}:${String(s%60).padStart(2,'0')}`;
  async function call(fn,args){const {data,error}=await supabase.rpc(fn,args);if(error)throw error;return data;}
  async function signIn(e){e.preventDefault();setBusy(true);setMsg('');const {error}=await supabase.auth.signInWithPassword({email:adminEmail,password:adminPassword});setBusy(false);if(error)setMsg(error.message);}
  async function logout(){await supabase.auth.signOut();setPage('home');}
  async function loadExams(){const {data,error}=await supabase.from('exams').select('*').order('created_at',{ascending:false});if(error){setMsg(error.message);return;}setExams(data||[]);}
  async function createExam(e){e.preventDefault();setBusy(true);setMsg('');const {data,error}=await supabase.from('exams').insert({...newExam,code:newExam.code.trim().toUpperCase(),created_by:session.user.id}).select().single();setBusy(false);if(error){setMsg(error.message);return;}setNewExam({title:'',code:'',duration_minutes:60});setEditingExam(data);setPage('manage');loadQuestions(data.id);}
  async function loadQuestions(id){const {data,error}=await supabase.from('questions').select('*').eq('exam_id',id).order('position');if(error){setMsg(error.message);return;}setExamQuestions(data||[]);}
  async function toggleExam(e){const {error}=await supabase.from('exams').update({is_active:!e.is_active}).eq('id',e.id);if(error)setMsg(error.message);else loadExams();}
  async function addQuestion(e){e.preventDefault();if(!editingExam)return;setBusy(true);let payload={exam_id:editingExam.id,type:draft.type,prompt:draft.prompt,points:Number(draft.points),position:examQuestions.length+1,options:draft.type==='multiple_choice'?draft.options.filter(x=>x.trim()):null,correct_answer:['multiple_choice','short_answer'].includes(draft.type)?draft.correct_answer.trim():null};const {error}=await supabase.from('questions').insert(payload);setBusy(false);if(error){setMsg(error.message);return;}setDraft({type:'multiple_choice',prompt:'',options:['','','',''],correct_answer:'',points:1});setMsg('Soal berhasil ditambahkan.');loadQuestions(editingExam.id);}
  async function deleteQuestion(id){if(!confirm('Hapus soal ini?'))return;const {error}=await supabase.from('questions').delete().eq('id',id);if(error)setMsg(error.message);else loadQuestions(editingExam.id);}
  async function startExam(e){e.preventDefault();setBusy(true);setMsg('');try{const data=await call('student_start_exam',{p_code:student.code.trim().toUpperCase(),p_name:student.name.trim(),p_class:student.class_name.trim()});setExam(data.exam);setQuestions(data.questions);setAnswers(data.answers||{});setRemaining(data.remaining_seconds);setIdx(0);setResult(null);setPage('taking');}catch(err){setMsg(err.message||'Tidak dapat memulai ujian.');}setBusy(false);}
  async function saveAnswer(qid,value){setAnswers(a=>({...a,[qid]:value}));try{await call('student_save_answer',{p_attempt_id:exam.attempt_id,p_question_id:qid,p_answer:value});}catch(err){setMsg('Jawaban belum tersimpan: '+err.message);}}
  async function finishExam(auto=false){if(!exam?.attempt_id)return;if(!auto&&!confirm('Yakin ingin mengumpulkan ujian?'))return;setBusy(true);try{const data=await call('student_finish_exam',{p_attempt_id:exam.attempt_id});setResult(data);setPage('done');}catch(err){setMsg(err.message||'Gagal mengumpulkan ujian.');}setBusy(false);}
  async function openGrades(e){setEditingExam(e);const {data,error}=await supabase.from('attempts').select('*').eq('exam_id',e.id).order('started_at',{ascending:false});if(error){setMsg(error.message);return;}setGradeRows(data||[]);setPage('grades');}
  async function gradeEssay(row,score){const {error}=await supabase.from('attempts').update({manual_score:Number(score)}).eq('id',row.id);if(error)setMsg(error.message);else openGrades(editingExam);}
  const answered=Object.keys(answers).filter(k=>String(answers[k]??'').trim()!=='').length;
  if(!configured)return <main className="shell"><Header/><div className="panel"><h2>Konfigurasi belum diisi</h2><p>Isi URL dan anon key Supabase pada file <code>.env</code>, lalu jalankan ulang aplikasi. Petunjuk lengkap ada di README.md.</p></div></main>;
  return <main className="shell"><Header/>
    {msg&&<div className="notice" role="status">{msg}<button className="x" onClick={()=>setMsg('')}>×</button></div>}
    {page==='home'&&<div className="grid two"><section className="panel"><span className="eyebrow">PESERTA DIDIK</span><h2>Masuk Ujian</h2><p>Masukkan identitas dan kode ujian dari guru.</p><form onSubmit={startExam}><Field label="Nama lengkap"><input required value={student.name} onChange={e=>setStudent({...student,name:e.target.value})}/></Field><Field label="Kelas"><input required placeholder="Contoh: VI A" value={student.class_name} onChange={e=>setStudent({...student,class_name:e.target.value})}/></Field><Field label="Kode ujian"><input required value={student.code} onChange={e=>setStudent({...student,code:e.target.value.toUpperCase()})}/></Field><button disabled={busy} className="primary wide">{busy?'Memproses...':'Mulai Ujian'}</button></form></section><section className="panel soft"><span className="eyebrow">PENGELOLA</span><h2>Login Guru / Admin</h2><p>Masuk untuk membuat ujian, mengelola soal, dan melihat hasil.</p><form onSubmit={signIn}><Field label="Email administrator"><input type="email" required value={adminEmail} onChange={e=>setAdminEmail(e.target.value)}/></Field><Field label="Kata sandi"><input type="password" required value={adminPassword} onChange={e=>setAdminPassword(e.target.value)}/></Field><button disabled={busy} className="secondary wide">Masuk Dashboard</button></form></section></div>}
    {page==='admin'&&<><div className="toolbar"><div><span className="eyebrow">DASHBOARD</span><h2>Kelola Ujian</h2></div><button className="ghost" onClick={logout}>Keluar</button></div><div className="grid two"><section className="panel"><h3>Buat ujian baru</h3><form onSubmit={createExam}><Field label="Nama ujian"><input required value={newExam.title} onChange={e=>setNewExam({...newExam,title:e.target.value})}/></Field><Field label="Kode ujian"><input required value={newExam.code} onChange={e=>setNewExam({...newExam,code:e.target.value.toUpperCase()})}/></Field><Field label="Durasi (menit)"><input type="number" min="1" max="300" required value={newExam.duration_minutes} onChange={e=>setNewExam({...newExam,duration_minutes:e.target.value})}/></Field><button className="primary" disabled={busy}>Buat ujian</button></form></section><section className="panel"><h3>Daftar ujian</h3>{exams.length===0?<p>Belum ada ujian.</p>:exams.map(e=><div className="examrow" key={e.id}><div><b>{e.title}</b><small>Kode: {e.code} · {e.duration_minutes} menit · {e.is_active?'Aktif':'Nonaktif'}</small></div><div className="actions"><button className="small" onClick={()=>{setEditingExam(e);loadQuestions(e.id);setPage('manage')}}>Soal</button><button className="small" onClick={()=>openGrades(e)}>Hasil</button><button className="small" onClick={()=>toggleExam(e)}>{e.is_active?'Tutup':'Aktifkan'}</button></div></div>)}</section></div></>}
    {page==='manage'&&editingExam&&<><div className="toolbar"><div><span className="eyebrow">PENGELOLAAN SOAL</span><h2>{editingExam.title}</h2><p>Kode ujian: <b>{editingExam.code}</b></p></div><button className="ghost" onClick={()=>setPage('admin')}>← Dashboard</button></div><div className="grid two"><section className="panel"><h3>Tambah soal</h3><form onSubmit={addQuestion}><Field label="Jenis soal"><select value={draft.type} onChange={e=>setDraft({...draft,type:e.target.value})}><option value="multiple_choice">Pilihan ganda</option><option value="short_answer">Isian singkat</option><option value="essay">Uraian</option></select></Field><Field label="Pertanyaan"><textarea required rows="4" value={draft.prompt} onChange={e=>setDraft({...draft,prompt:e.target.value})}/></Field>{draft.type==='multiple_choice'&&<><Field label="Pilihan jawaban (isi minimal 2)">{draft.options.map((o,i)=><input key={i} className="optioninput" placeholder={`Pilihan ${String.fromCharCode(65+i)}`} value={o} onChange={e=>setDraft({...draft,options:draft.options.map((x,j)=>j===i?e.target.value:x)})}/>)}</Field><Field label="Kunci jawaban (tulis persis seperti pilihan)"><input required value={draft.correct_answer} onChange={e=>setDraft({...draft,correct_answer:e.target.value})}/></Field></>}{draft.type==='short_answer'&&<Field label="Kunci jawaban"><input required value={draft.correct_answer} onChange={e=>setDraft({...draft,correct_answer:e.target.value})}/></Field>}<Field label="Bobot nilai"><input type="number" min="0" step="0.5" value={draft.points} onChange={e=>setDraft({...draft,points:e.target.value})}/></Field><button className="primary" disabled={busy}>Simpan soal</button></form></section><section className="panel"><h3>Soal tersimpan ({examQuestions.length})</h3>{examQuestions.map((q,i)=><div className="questionrow" key={q.id}><div><b>{i+1}. {q.prompt}</b><small>{q.type==='multiple_choice'?'Pilihan ganda':q.type==='short_answer'?'Isian singkat':'Uraian'} · {q.points} poin</small></div><button className="danger small" onClick={()=>deleteQuestion(q.id)}>Hapus</button></div>)}</section></div></>}
    {page==='grades'&&<><div className="toolbar"><div><span className="eyebrow">HASIL UJIAN</span><h2>{editingExam?.title}</h2></div><button className="ghost" onClick={()=>setPage('admin')}>← Dashboard</button></div><section className="panel"><div className="tablewrap"><table><thead><tr><th>Nama</th><th>Kelas</th><th>Mulai</th><th>Status</th><th>Nilai objektif</th><th>Nilai uraian</th></tr></thead><tbody>{gradeRows.map(r=><tr key={r.id}><td>{r.student_name}</td><td>{r.class_name}</td><td>{new Date(r.started_at).toLocaleString('id-ID')}</td><td>{r.status}</td><td>{r.objective_score??'—'}</td><td><input className="score" type="number" min="0" max="100" placeholder="Nilai" defaultValue={r.manual_score??''} onBlur={e=>gradeEssay(r,e.target.value)}/></td></tr>)}</tbody></table></div><p className="muted">Nilai uraian diisi guru secara manual. Ekspor CSV dapat ditambahkan pada tahap berikutnya.</p></section></>}
    {page==='taking'&&current&&<><div className="examhead"><div><span className="eyebrow">{exam.title}</span><h2>{student.name} · {student.class_name}</h2></div><div className="timer"><small>SISA WAKTU</small><b>{fmt(remaining)}</b></div></div><div className="examlayout"><aside className="panel navpanel"><b>Nomor soal</b><div className="numbergrid">{questions.map((q,i)=><button key={q.id} className={`number ${i===idx?'selected':''} ${String(answers[q.id]??'').trim()!==''?'answered':''}`} onClick={()=>setIdx(i)}>{i+1}</button>)}</div><small>{answered} dari {questions.length} terjawab</small></aside><section className="panel questionpanel"><div className="qmeta">SOAL {idx+1} DARI {questions.length} · {current.points} POIN</div><h3>{current.prompt}</h3>{current.type==='multiple_choice'?<div className="choices">{(current.options||[]).map((o,i)=><label className={`choice ${answers[current.id]===o?'chosen':''}`} key={i}><input type="radio" name={current.id} checked={answers[current.id]===o} onChange={()=>saveAnswer(current.id,o)}/><span>{String.fromCharCode(65+i)}.</span>{o}</label>)}</div>:<textarea rows="6" placeholder={current.type==='essay'?'Tuliskan jawaban uraian...':'Tuliskan jawaban...'} value={answers[current.id]||''} onChange={e=>saveAnswer(current.id,e.target.value)}/>}<div className="qactions"><button className="ghost" disabled={idx===0} onClick={()=>setIdx(idx-1)}>Sebelumnya</button><button className="secondary" disabled={idx===questions.length-1} onClick={()=>setIdx(idx+1)}>Berikutnya</button></div><button className="primary wide submit" onClick={()=>finishExam(false)} disabled={busy}>Kumpulkan ujian</button></section></div></>}
    {page==='done'&&<section className="panel centered"><div className="check">✓</div><h2>Ujian berhasil dikumpulkan</h2><p>Terima kasih, {student.name}. Jawaban Anda telah disimpan.</p>{result?.objective_score!=null&&<p>Nilai objektif: <b>{result.objective_score}</b></p>}<button className="primary" onClick={()=>{setPage('home');setStudent({name:'',class_name:'',code:''});setMsg('');}}>Kembali ke halaman awal</button></section>}
    <footer>© {new Date().getFullYear()} {SCHOOL_NAME} · CBT Sekolah</footer>
  </main>;
}
function Header(){return <header className="brand">{SCHOOL_LOGO_URL?<img src={SCHOOL_LOGO_URL} alt="Logo sekolah"/>:<div className="brandmark">CBT</div>}<div><b>{SCHOOL_NAME}</b><small>Computer Based Test · Ujian Online</small></div></header>}
function Field({label,children}){return <label className="field"><span>{label}</span>{children}</label>}
createRoot(document.getElementById('root')).render(<App/>);
